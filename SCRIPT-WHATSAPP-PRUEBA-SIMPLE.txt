📱 SCRIPT PYTHON SIMPLE - PRUEBA WHATSAPP (Solo 1 número)
========================================================================================

Este script envía UN mensaje a UN número solamente.
Perfecto para probar que todo funciona.

========================================================================================
PASO 1: COPIA ESTE CÓDIGO COMPLETO
========================================================================================

import requests

# TUS CREDENCIALES ZADARMA
ZADARMA_API_KEY = "1f2c9daf0db1307a1128"
ZADARMA_API_SECRET = "a53586ef23bcdd1e220e"

# NÚMERO DONDE ENVIAR (la prueba)
NUMERO_PRUEBA = "+5491160206752"

# TU INFORMACIÓN
TU_NOMBRE = "Juan García"                    # CAMBIA ESTO
TU_TELEFONO = "+54 9 351 555-6666"          # CAMBIA ESTO

# EL MENSAJE
MENSAJE = f"""Hola, soy {TU_NOMBRE}.

Ofrezco servicios de IA para clínicas dentales:
✅ Chatbots para agendamiento 24/7
✅ CRM para seguimiento de pacientes
✅ Email marketing automático

¿Podemos hablar 15 minutos?

Teléfono: {TU_TELEFONO}"""

# ENVÍA EL MENSAJE
url = "https://api.zadarma.com/v1/sms/send"

params = {
    "api_key": ZADARMA_API_KEY,
    "api_secret": ZADARMA_API_SECRET,
    "to": NUMERO_PRUEBA,
    "message": MENSAJE
}

print("📱 Enviando mensaje de prueba...")
print(f"📞 Número: {NUMERO_PRUEBA}")
print(f"💬 Mensaje:\n{MENSAJE}\n")

try:
    response = requests.post(url, data=params)
    data = response.json()

    if data.get("status") == 0:
        print("✅ ¡MENSAJE ENVIADO EXITOSAMENTE!")
        print(f"ID: {data.get('message_id')}")
    else:
        print(f"❌ Error: {data.get('message')}")

except Exception as e:
    print(f"❌ Error de conexión: {e}")
