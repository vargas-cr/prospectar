import csv
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from datetime import datetime

# ===== TUS DATOS =====
TU_EMAIL = "tu_email@gmail.com"          # TU EMAIL (desde donde envías)
TU_PASSWORD = "tu_app_password"          # TU APP PASSWORD de Gmail
TU_NOMBRE = "Tu Nombre"                  # Tu nombre
TU_TELÉFONO = "Tu Teléfono"             # Tu teléfono

# ===== CONFIGURACIÓN GMAIL =====
SMTP_SERVER = "smtp.gmail.com"
SMTP_PORT = 587

def enviar_email(email_destino, nombre_negocio, asunto, cuerpo):
    """Envía email personalizado"""

    try:
        # Conecta a Gmail
        server = smtplib.SMTP(SMTP_SERVER, SMTP_PORT)
        server.starttls()
        server.login(TU_EMAIL, TU_PASSWORD)

        # Crea el email
        mensaje = MIMEMultipart()
        mensaje['From'] = TU_EMAIL
        mensaje['To'] = email_destino
        mensaje['Subject'] = asunto

        # Personaliza el cuerpo
        cuerpo_final = cuerpo.replace("[TU NOMBRE]", TU_NOMBRE)
        cuerpo_final = cuerpo_final.replace("[TU TELÉFONO]", TU_TELÉFONO)
        cuerpo_final = cuerpo_final.replace("[NOMBRE_NEGOCIO]", nombre_negocio)

        mensaje.attach(MIMEText(cuerpo_final, 'plain'))

        # Envía
        server.send_message(mensaje)
        server.quit()

        print(f"✅ Email enviado a {email_destino} ({nombre_negocio})")
        return True

    except Exception as e:
        print(f"❌ Error enviando a {email_destino}: {e}")
        return False

def procesar_csv(archivo_csv):
    """Lee el CSV y envía emails"""

    enviados = 0
    fallidos = 0

    try:
        with open(archivo_csv, 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)

            for row in reader:
                nombre_negocio = row.get('Nombre Negocio')
                email = row.get('Email').strip()
                zona = row.get('Zona')
                estado = row.get('Estado Envío', '').strip()

                # Solo envía si Status está vacío
                if not estado or estado == "":

                    # Asunto personalizado
                    asunto = f"{nombre_negocio}: Estás perdiendo clientes por no estar en Google"

                    # Cuerpo personalizado (ejemplo)
                    cuerpo = f"""Hola,

Vi que {nombre_negocio} hace excelente trabajo eléctrico en {zona}.

Pero hay un problema: NO tienen sitio web.

¿Qué significa eso?

AHORA (sin web):
- Clientes buscan "Electricista {zona}" en Google
- Encuentran competencia CON web
- Van con ellos
- Ustedes pierden ese cliente

NÚMEROS:
- {zona}: ~400 búsquedas/mes de "Electricista"
- Sin web pierden: ~$300,000-400,000/AÑO

¿Hablamos 15 minutos sobre cómo aparecer en Google?

[TU NOMBRE]
[TU TELÉFONO]"""

                    # Envía
                    if enviar_email(email, nombre_negocio, asunto, cuerpo):
                        enviados += 1
                        # Aquí podrías actualizar el CSV con "Enviado"
                    else:
                        fallidos += 1

    except FileNotFoundError:
        print(f"❌ No encontró el archivo: {archivo_csv}")
        return

    print(f"\n{'='*50}")
    print(f"✅ Enviados: {enviados}")
    print(f"❌ Fallidos: {fallidos}")
    print(f"{'='*50}")

# ===== EJECUTA =====
if __name__ == "__main__":
    print("📧 ENVÍO MASIVO A ELECTRICISTAS")
    print("="*50)

    # Nombre del archivo CSV
    archivo = "electricistas_ba.csv"

    # Verifica que tengas credenciales
    if TU_EMAIL == "tu_email@gmail.com":
        print("❌ CONFIGURA TU EMAIL PRIMERO")
        print("Reemplaza: TU_EMAIL y TU_PASSWORD")
        exit()

    # Procesa
    procesar_csv(archivo)

========================================================================================
TROUBLESHOOTING
========================================================================================

❌ "Error: Login failed"
→ Tu contraseña de aplicación es incorrecta
→ Genera una nueva en myaccount.google.com

❌ "Error: File not found"
→ El CSV no está en la misma carpeta que el script
→ Mueve electricistas_ba.csv a la carpeta del script

❌ "Error: Invalid email address"
→ Un email en el CSV está mal escrito
→ Verifica que sea correcto

❌ "ModuleNotFoundError"
→ Instala: pip install secure-smtplib
