def enviar_mensaje_whatsapp(numero, mensaje):
    """Envía mensaje por WhatsApp usando Zadarma API"""

    url = "https://api.zadarma.com/v1/sms/send"

    params = {
        "api_key": ZADARMA_API_KEY,
        "api_secret": ZADARMA_API_SECRET,
        "to": numero,
        "message": mensaje,
        "phone_id": ZADARMA_PHONE  # Tu número de Zadarma
    }

    try:
        response = requests.post(url, data=params)
        data = response.json()

        if data.get("status") == 0:  # 0 = éxito
            print(f"✅ Mensaje enviado a {numero}")
            return True
        else:
            print(f"❌ Error: {data.get('message')}")
            return False

    except Exception as e:
        print(f"❌ Error enviando: {e}")
        return False

def leer_google_sheet(sheet_id):
    """Lee datos del Google Sheet"""

    print("\n1. Ve a: https://docs.google.com/spreadsheets/d/{}/export?format=csv".format(sheet_id))

    input()

    leads = []

    try:

def enviar_a_todos_los_leads(leads):
    """Envía mensaje a todos los leads"""

    enviados = 0
    fallidos = 0

    for i, lead in enumerate(leads):
        clinica = lead.get('Nombre Clínica', 'N/A')
        numero = lead.get('Teléfono', '').strip()
        estado = lead.get('Estado', '').strip()
        mensaje_personalizado = lead.get('Mensaje WhatsApp', '')

        # Solo envía si Status está vacío
        if not estado or estado == "":

            if not numero:
                print(f"❌ {clinica}: Sin teléfono")
                fallidos += 1
                continue

            # Personaliza el mensaje
            mensaje = mensaje_personalizado.replace("[TU NOMBRE]", TU_NOMBRE)
            mensaje = mensaje.replace("[TU TELÉFONO]", TU_TELEFONO)

            # Asegúrate que el número esté en formato internacional
            if not numero.startswith("+"):
                numero = "+54" + numero.lstrip("0")

            # Envía
            print(f"\n📱 Enviando a {clinica} ({numero})...")
            if enviar_mensaje_whatsapp(numero, mensaje):
                enviados += 1
                # Aquí actualizarías el Sheet, pero es manual por ahora
            else:
                fallidos += 1

            # Pausa entre mensajes (para no saturar)
            import time
            time.sleep(2)

    print(f"\n{'='*50}")
    print(f"✅ Enviados: {enviados}")
    print(f"❌ Fallidos: {fallidos}")
    print(f"{'='*50}")
