function enviarEmails() {
  // Tu información
  const TU_NOMBRE = "Juan García";                    // CAMBIA ESTO
  const TU_TELEFONO = "+54 9 351 555-6666";           // CAMBIA ESTO
  const TU_EMAIL = "tu@email.com";                    // CAMBIA ESTO
  const TU_GMAIL = "argentia.ai@gmail.com";           // Tu Gmail

  // Lee el Sheet
  const sheet = SpreadsheetApp.getActiveSheet();
  const data = sheet.getDataRange().getValues();

  // Envía email por cada clínica
  for (let i = 1; i < data.length; i++) {  // Empieza en fila 2
    const clinica = data[i][0];             // Columna A: Nombre Clínica
    const email = data[i][1];               // Columna B: Email
    const estado = data[i][6];              // Columna G: Estado

    // Si Status está vacío, envía email
    if (!estado || estado.trim() === "") {
      const asunto = "Oportunidad IA para " + clinica;
      const cuerpo = data[i][5];            // Columna F: Email personalizado

      // Personaliza el email
      let emailFinal = cuerpo
        .replace(/\[TU NOMBRE\]/g, TU_NOMBRE)
        .replace(/\[TU TELÉFONO\]/g, TU_TELEFONO)
        .replace(/\[TU EMAIL\]/g, TU_EMAIL);

      // Envía
      try {
        GmailApp.sendEmail(email, asunto, emailFinal, {
          from: TU_GMAIL,
          name: TU_NOMBRE,
          replyTo: TU_EMAIL
        });

        // Marca como "Enviado"
        sheet.getRange(i + 1, 7).setValue("Enviado");
        sheet.getRange(i + 1, 8).setValue(new Date());

        Logger.log("✅ Email enviado a: " + email);

      } catch (e) {
        Logger.log("❌ Error: " + email + " - " + e);
      }
    }
  }

  Logger.log("✅ TODOS LOS EMAILS FUERON ENVIADOS");
}

========================================================================================
PARA AUTOMATIZAR (EJECUTAR CADA DÍA SIN HACER NADA)
========================================================================================

Si quieres que se ejecute AUTOMÁTICAMENTE cada día:

1. En Google Apps Script, click en ⏱️ ACTIVADORES (lado izquierdo)
2. Click en "+ CREAR ACTIVADOR"
3. Configura así:
   - Función: "enviarEmails"
   - Tipo de evento: "Evento en tiempo"
   - Intervalo: "Cada día"
   - Hora: "10:00 AM" (la hora que quieras)
4. Click GUARDAR

AHORA: Cada día a las 10am, se envían automáticamente todos los emails pendientes.

TROUBLESHOOTING RÁPIDO
========================================================================================

❌ "Error: Permission denied"
→ Autoriza en la primera ejecución
→ Es normal, solo la primera vez

❌ "No veo nada en el Estado"
→ Refresca el Sheet (F5)
→ Espera 5-10 segundos

❌ "Dice error en un email"
→ Ese email está mal escrito en Columna B
→ Corrígelo y ejecuta de nuevo

❌ "Quiero agregar más clínicas"
→ Agrega fila nueva en el Sheet
→ Llena: Nombre, Email, etc.
→ Columna G déjala vacía
→ Ejecuta nuevamente
